FLOORED_NODES = 224


def floored(raw, m):
    """The north-band grid as the City draws it before this patch: the shared
    Connector floors of north-band-floors.json, applied exactly as
    web/src/square-board/connector-ground.ts applies them (int16, JS rounding,
    the lowest floor wins). Blending over this grid instead of the raw DEM keeps
    the patch from restoring the deck and roof returns those floors removed."""
    spec = json.loads((HERE / "north-band-floors.json").read_text())
    if m["sha256"] != spec["groundSha256"]:
        return raw
    out = raw.astype(np.int64).copy()
    step, b = m["stepM"], m["bounds"]
    for f in spec["floors"]:
        (x0, z0, h0), (x1, z1, h1), width, crossfall = f["start"], f["end"], f["halfWidthM"], f["crossfall"]
        dx, dz = x1 - x0, z1 - z0
        length, pad = math.hypot(dx, dz), width + step
        row0 = max(1, math.floor((min(z0, z1) - pad - b["minZ"]) / step))
        row1 = min(m["rows"] - 2, math.ceil((max(z0, z1) + pad - b["minZ"]) / step))
        col0 = max(1, math.floor((min(x0, x1) - pad - b["minX"]) / step))
        col1 = min(m["cols"] - 2, math.ceil((max(x0, x1) + pad - b["minX"]) / step))
        for row in range(row0, row1 + 1):
            for col in range(col0, col1 + 1):
                x, z = b["minX"] + col * step - x0, b["minZ"] + row * step - z0
                t = (x * dx + z * dz) / (length * length)
                side = (z * dx - x * dz) / length
                if t < 0 or t > 1 or abs(side) >= pad or raw[row, col] == m["noData"]:
                    continue
                old = m["offsetM"] + m["scaleM"] * float(raw[row, col])
                floor = h0 + (h1 - h0) * t + crossfall * side
                if old - floor < 0.45:
                    continue
                weight = min(1, (pad - abs(side)) / step)
                lower = math.floor((old - weight * (old - floor) - m["offsetM"]) / m["scaleM"] + 0.5)
                out[row, col] = min(out[row, col], lower)
    # Shared endpoints: nodes on the outside of the bend lie beyond both
    # segments. Give them the joint's floor (connector-ground.ts does the same).
    def covers(f, x, z):
        (x0, z0, _), (x1, z1, _) = f["start"], f["end"]
        dx, dz = x1 - x0, z1 - z0
        length = math.hypot(dx, dz)
        t = ((x - x0) * dx + (z - z0) * dz) / (length * length)
        side = ((z - z0) * dx - (x - x0) * dz) / length
        return 0 <= t <= 1 and abs(side) < f["halfWidthM"] + step
    floors = spec["floors"]
    for i, a in enumerate(floors):
        for b2 in floors[i + 1:]:
            joint = next((p for p in (a["start"], a["end"]) if any(p[0] == q[0] and p[1] == q[1] for q in (b2["start"], b2["end"]))), None)
            if joint is None:
                continue
            jx, jz, jh = joint
            pad = min(a["halfWidthM"], b2["halfWidthM"]) + step
            for row in range(max(1, math.floor((jz - pad - b["minZ"]) / step)), min(m["rows"] - 2, math.ceil((jz + pad - b["minZ"]) / step)) + 1):
                for col in range(max(1, math.floor((jx - pad - b["minX"]) / step)), min(m["cols"] - 2, math.ceil((jx + pad - b["minX"]) / step)) + 1):
                    x, z = b["minX"] + col * step, b["minZ"] + row * step
                    distance = math.hypot(x - jx, z - jz)
                    if distance >= pad or any(covers(f, x, z) for f in floors) or raw[row, col] == m["noData"]:
                        continue
                    old = m["offsetM"] + m["scaleM"] * float(raw[row, col])
                    if old - jh < 0.45:
                        continue
                    weight = min(1, (pad - distance) / step)
                    out[row, col] = min(out[row, col], math.floor((old - weight * (old - jh) - m["offsetM"]) / m["scaleM"] + 0.5))
    changed = int((out != raw).sum())
    if changed != FLOORED_NODES:
        raise ValueError(f"north-band floors changed {changed} nodes, connector-ground.test.ts pins {FLOORED_NODES}")
    return out


# Ramps that ride above the covered lower roads (the MLK SB ramps climbing to the
# deck on their own structure, the Baker exit continuing flyover 38974028 at
# street level). Their lidar centreline sits on the raw grid, metres above the
# lower-road plane, so the taper must leave their lanes alone.
lower_ways = {wid for rows in sections.values() for wid, *_ in rows}
other_ways = [
    (int(wid), np.array(way["pts"], dtype=float), way["lanes"] * 3.55 / 2 + 1.5)
    for wid, way in json.loads((HERE / "ways.json").read_text()).items()
    if int(wid) not in lower_ways
]

surveys = json.loads((HERE / "junction-decks.json").read_text()) + json.loads(
    (HERE / "street-bridges.json").read_text()
)
planes = []
outlines = []
ground_returns = []
for site in surveys:
    if "rows" in site:
        sid = site["outline"]
        rows = site["rows"]
        outline = Polygon([r[:2] for r in rows] + [r[3:5] for r in reversed(rows)])
    else:
        sid = site["outlineId"]
        outline = Polygon(site["outline"])
    xmin, zmin, xmax, zmax = outline.buffer(55).bounds
    lon = lambda x: LON0 + math.degrees(x / (R * K))
    lat = lambda z: LAT0 - math.degrees(z / R)
    x, z, h, c, _ = ept_points((lon(xmin), lat(zmax), lon(xmax), lat(zmin)))
    exposed = (c == 2) & ~contains_xy(outline.buffer(3), x, z)
    outlines.append(outline.buffer(3))
    ground_returns.append(np.stack([x[c == 2], z[c == 2], h[c == 2]], 1))
    for wid, ax, az, bx, bz, width in sections[sid]:
        length = math.hypot(bx - ax, bz - az)
        dx, dz = (bx - ax) / length, (bz - az) / length
        u = (x - ax) * dx + (z - az) * dz
        v = (z - az) * dx - (x - ax) * dz
        ends = ((u > -32) & (u < -4)) | ((u > length + 4) & (u < length + 32))
        use = exposed & ends & (abs(v) < width)
        for side in [u < 0, u > length]:
            sample = h[use & side]
            if len(sample) < 12:
                raise ValueError((wid, len(sample)))
            use &= ~side | (abs(h - np.median(sample)) < 1.0)
        design = np.stack([u, v, np.ones(len(u))], 1)
        for _ in range(3):
            plane = np.linalg.lstsq(design[use], h[use], rcond=None)[0]
            use &= abs(h - design @ plane) < 0.45
        if abs(plane[1]) > 0.09:
            raise ValueError((wid, plane))
        planes.append((ax, az, dx, dz, length, width, plane))
        print(wid, sum(use), plane.round(4), flush=True)

patches = {}
for source in ["downtown-ground", "square-board/north-band", "square-board/south-band"]:
    directory = HERE.parent / source
    m = json.loads((directory / "ground.json").read_text())
    raw = np.fromfile(directory / "ground.i16", dtype="<i2").reshape(
        m["rows"], m["cols"]
    )
    # Blend over the grid the renderer actually holds at this stage. Indices are
    # written where the result differs from that grid, and nodeHeights applies
    # them after the grid corrections.
    raw = floored(raw, m)
    old = raw * m["scaleM"] + m["offsetM"]
    j, i = np.indices(old.shape)
    x = m["bounds"]["minX"] + i * m["stepM"]
    z = m["bounds"]["minZ"] + j * m["stepM"]
    cover = np.zeros_like(old)
    total = cover.copy()
    weighted = cover.copy()
    for ax, az, dx, dz, length, width, plane in planes:
        s = (x - ax) * dx + (z - az) * dz
        t = (z - az) * dx - (x - ax) * dz
        w = np.clip(
            np.minimum.reduce(
                [(width + 20 - abs(t)) / 9, (s + 44) / 16, (length + 44 - s) / 16]
            ),
            0,
            1,
        )
        cover = np.maximum(cover, w)
        weight = w * np.exp(-0.5 * (t / (0.65 * width + 4)) ** 2)
        total += weight
        weighted += weight * (plane[0] * s + plane[1] * t + plane[2])
    # Tie into the unchanged board-edge nodes before the shared mosaic seam.
    edge = np.minimum.reduce([i, j, m["cols"] - 1 - i, m["rows"] - 1 - j])
    cover *= np.clip(edge / 2, 0, 1)
    lower = weighted / np.maximum(total, 1e-12)
    from scipy.spatial import cKDTree
    ground_xyz = np.concatenate(ground_returns)
    ground_tree = cKDTree(ground_xyz[:, :2])
    # Under a surveyed deck the raw grid is the deck itself; the patch owns those
    # nodes for the lower road whatever way is mapped there.
    under_deck = np.zeros(old.shape, bool)
    for o in outlines:
        under_deck |= contains_xy(o, x, z)
    for _, pts, half in other_ways:
        a, b = pts[:-1], pts[1:]
        if ((x < pts[:, 0].min() - half - 5) | (x > pts[:, 0].max() + half + 5)).all():
            continue
        near = np.zeros(old.shape, bool)
        for (ax, az), (bx, bz) in zip(a, b):
            vx, vz = bx - ax, bz - az
            t = np.clip(((x - ax) * vx + (z - az) * vz) / max(vx * vx + vz * vz, 1e-9), 0, 1)
            near |= np.hypot(x - ax - t * vx, z - az - t * vz) < half
        for jj, ii in zip(*np.nonzero(near & ~under_deck & (cover > 0) & (old - lower > 1.5))):
            # Only where the way's own class-2 returns confirm the raw grid is
            # its surface; a raw node over contaminated ground keeps the patch.
            hits = ground_tree.query_ball_point([x[jj, ii], z[jj, ii]], 2.5)
            if len(hits) >= 3 and abs(np.median(ground_xyz[hits, 2]) - old[jj, ii]) < 1.0:
                cover[jj, ii] = 0
    result = old + cover * (lower - old)
    q = np.round(result, 2)
    indices = np.flatnonzero((abs(q - old) > 0.015) & (raw != m["noData"]))
    patches[m["sha256"]] = [[int(i), float(q.flat[i])] for i in indices]
(HERE / "underpass-floors.json").write_text(
    json.dumps(patches, separators=(",", ":")) + "\n"
)
