import { readFileSync } from 'node:fs';
import { it } from 'vitest';
import survey from '../../pipeline/prototypes/connector-footprint/ramp-decks.json';
import { auburnTransport } from './city/sweet-auburn-transport';
import { cityRoadHeight } from './city/road-height';
import { GroundGrid } from './terrain/grid';
import { applyRampDecks } from './terrain/ramp-decks';
import { GroundSurface } from './terrain/surface';
import { createSurfaceTransport } from './terrain/transport';
import type { TerrainCity } from './terrain/transport-data';
function load(dir: string, ctx: string) {
  const m = JSON.parse(readFileSync(dir + 'ground.json', 'utf8'));
  return { m, grid: new GroundGrid(m, Uint8Array.from(readFileSync(dir + 'ground.i16')).buffer), city: JSON.parse(readFileSync(dir + ctx, 'utf8')) as TerrainCity };
}
function seg(p: number[], a: number[], b: number[]) { const dx=b[0]-a[0],dz=b[1]-a[1],t=Math.max(0,Math.min(1,((p[0]-a[0])*dx+(p[1]-a[1])*dz)/(dx*dx+dz*dz))); return Math.hypot(p[0]-a[0]-t*dx,p[1]-a[1]-t*dz); }
it.each(['low','high'] as const)('cars %s', q => {
  for (const [name, dir, ctx, prep, inside] of [
    ['downtown', 'public/data/scene/downtown/', 'vignette.json', (c: TerrainCity) => applyRampDecks(c), (x: number) => x < 1000],
    ['north-corner', '../pipeline/prototypes/square-board/north-corner/', 'context.json', (c: TerrainCity) => auburnTransport(c), (x: number) => x > 1000],
  ] as const) {
    const { m, grid, city: raw } = load(dir, ctx);
    const city = prep(raw);
    const ground = new GroundSurface(grid, q);
    const gAt = ground.sampleScene.bind(ground);
    const t = createSurfaceTransport(city, grid, gAt);
    const height = cityRoadHeight(city, (x, z) => gAt(x, z), t.deckAt);
    const zero = m.vertical.sceneZeroNavd88M;
    for (const w of survey.ways) {
      let worst = .12, n = 0, nul = 0, at = '';
      let run = 0;
      for (let i = 1; i < w.pts.length; i++) {
        const [a, b] = [w.pts[i-1], w.pts[i]], l = Math.hypot(b[0]-a[0], b[1]-a[1]);
        for (let s = 0; s < l; s += 1) {
          const st = run + s, sp = w.spans.find(p => st >= p[0][0] + .5 && st <= p.at(-1)![0] - .5);
          if (!sp) continue;
          const x = a[0] + (b[0]-a[0])*s/l, z = a[1] + (b[1]-a[1])*s/l;
          if (!inside(x)) continue;
          const road = city.roads.find(r => r.cls === 'motorway' && r.pts.some((p, k) => k && seg([x, z], r.pts[k-1], p) < .05) && r.pts.filter(p => w.pts.some((q, j) => j && seg(p, w.pts[j-1], q) < .05)).length >= 2);
          if (!road) { nul++; continue; }
          let k = 1; while (k < sp.length-1 && sp[k][0] < st) k++;
          const h = sp[k-1][1] + (sp[k][1]-sp[k-1][1])*(st-sp[k-1][0])/((sp[k][0]-sp[k-1][0])||1);
          const y = height(x, z, road as never);
          if (y == null) { nul++; continue; }
          const d = y + zero - h; n++;
          if (Math.abs(d - .12) > Math.abs(worst - .12)) { worst = d; at = `${st.toFixed(0)}`; }
        }
        run += l;
      }
      if (n || nul) console.log('CAR', q, name, w.way, 'samples', n, 'null/no-road', nul, 'worst car-profile', worst.toFixed(3), '@', at);
    }
    t.dispose(); ground.geometry.dispose();
  }
});
